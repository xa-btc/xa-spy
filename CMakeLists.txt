cmake_minimum_required(VERSION 3.24)
project(XASpy VERSION 0.2.0 LANGUAGES CXX)
# Include the MSVC runtime so portable copies need no VC++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "XA Spy currently supports Windows only")
endif()
enable_language(RC)
add_library(XASpyCore STATIC native/scanner.cpp native/windows_collectors.cpp native/browser.cpp)
target_compile_features(XASpyCore PUBLIC cxx_std_20)
target_compile_definitions(XASpyCore PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(XASpyCore PUBLIC bcrypt ole32 shell32 uuid advapi32 taskschd oleaut32 wbemuuid wevtapi iphlpapi ws2_32 setupapi cfgmgr32)
add_executable(XASpy WIN32 native/main.cpp native/theme.cpp native/resources.rc)
target_include_directories(XASpy PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets")
set_property(SOURCE native/resources.rc APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/xa-spy.ico")
target_compile_features(XASpy PRIVATE cxx_std_20)
target_compile_definitions(XASpy PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(XASpy PRIVATE XASpyCore comctl32 dwmapi uxtheme)
if(MSVC)
  target_compile_options(XASpy PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_compile_options(XASpyCore PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()
add_custom_command(TARGET XASpy POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/iocs" "$<TARGET_FILE_DIR:XASpy>/iocs")
option(XA_SPY_BUILD_TESTS "Build synthetic native privacy fixtures" OFF)
if(XA_SPY_BUILD_TESTS)
  enable_testing()
  target_compile_definitions(XASpyCore PRIVATE XA_SPY_TESTING)
  add_executable(XASpyPrivacyTests native/tests/privacy.cpp)
  target_compile_features(XASpyPrivacyTests PRIVATE cxx_std_20)
  target_compile_definitions(XASpyPrivacyTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(XASpyPrivacyTests PRIVATE XASpyCore)
  add_test(NAME native_privacy COMMAND XASpyPrivacyTests)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/native/tests/browser.cpp")
    add_executable(XASpyBrowserTests native/tests/browser.cpp)
    target_link_libraries(XASpyBrowserTests PRIVATE XASpyCore)
    add_test(NAME native_browser COMMAND XASpyBrowserTests)
  endif()
endif()
